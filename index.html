const SHEET_NAME = 'Data_PO'; function getOrCreateSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { const allSheets = ss.getSheets(); if (allSheets.length === 1 && allSheets[0].getLastRow() <= 1 && allSheets[0].getLastColumn() <= 1) { sheet = allSheets[0]; sheet.setName(SHEET_NAME); } else { sheet = ss.insertSheet(SHEET_NAME); } } if (sheet.getLastRow() === 0) { sheet.appendRow(['ID', 'Timestamp', 'Nama Pemesan', 'No. WhatsApp / HP', 'Alamat / Kota', 'Luar Kota', 'Jumlah Unit', 'Catatan / Request', 'Status Bayar', 'Total Biaya (Rp)']); sheet.getRange(1, 1, 1, 10).setFontWeight('bold').setBackground('#10b981').setFontColor('#ffffff'); sheet.setFrozenRows(1); } return sheet; } function doGet(e) { try { const action = (e.parameter && e.parameter.action) || 'getAll'; if (action === 'getAll') { const sheet = getOrCreateSheet(); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return createResponse({ status: 'success', count: 0, data: [] }); const data = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (!row[0] && !row[2]) continue; data.push({ id: String(row[0] || ''), timestamp: row[1] ? Utilities.formatDate(new Date(row[1]), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss') : '', nama: String(row[2] || ''), noHp: String(row[3] || ''), alamat: String(row[4] || ''), luarKota: row[5] === true || String(row[5]).toLowerCase() === 'ya' || String(row[5]).toLowerCase() === 'true', jumlahUnit: Number(row[6]) || 1, catatan: String(row[7] || ''), statusBayar: String(row[8] || 'Belum Bayar'), totalBiaya: Number(row[9]) || 0 }); } return createResponse({ status: 'success', count: data.length, data: data }); } } catch(err) { return createResponse({ status: 'error', message: err.toString() }); } } function doPost(e) { try { let payload = {}; if (e.postData && e.postData.contents) { try { payload = JSON.parse(e.postData.contents); } catch(pErr) { payload = e.parameter; } } else { payload = e.parameter; } const action = payload.action || 'add'; const sheet = getOrCreateSheet(); if (action === 'add') { const id = payload.id || 'PO-' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd-HHmmss'); sheet.appendRow([id, new Date(), payload.nama, payload.noHp, payload.alamat, payload.luarKota ? 'Ya' : 'Tidak', Number(payload.jumlahUnit)||1, payload.catatan||'', payload.statusBayar||'Belum Bayar', Number(payload.totalBiaya)||0]); return createResponse({ status: 'success', message: 'Tersimpan' }); } else if (action === 'updateStatus') { const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(payload.id)) { sheet.getRange(i + 1, 9).setValue(payload.statusBayar); return createResponse({ status: 'success' }); } } } else if (action === 'delete') { const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(payload.id)) { sheet.deleteRow(i + 1); return createResponse({ status: 'success' }); } } } return createResponse({ status: 'error', message: 'Aksi tidak dikenal' }); } catch(err) { return createResponse({ status: 'error', message: err.toString() }); } } function createResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
